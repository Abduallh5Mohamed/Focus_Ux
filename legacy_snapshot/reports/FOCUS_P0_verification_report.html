تأخير 5 ملّي ثانية بس بيحوّل +20 dB لـ -5.7 dB والسكريبت بيقص من الأول بس (n = min(len(ref), len(mix), len(est)))، يعني أي تأخير في مخرج الموديل بيتحوّل خطأ كامل. والموديل فعلاً بيأخّر: إعادة ترميز الصوت + نافذة 4 ثواني. قبل أي رقم، لازم تعملوا محاذاة (alignment) بالـcross-correlation، وتسجّلوا الإزاحة في الجدول. والبلان بيقول «اسمعوا المقطع قبل ما تثقوا في الرقم» — ده مش كفاية؛ الإزاحة مش مسموعة، هي 5 ملّي. 
# محاذاة قبل القياس (زوّدوا ده في أول السكريبت)
import torch, torch.nn.functional as F
def align(est, ref, max_lag=4000):          # ±250 ms
    x = F.conv1d(ref[None,None,:], est[None,None,:], padding=max_lag)[0,0]
    lag = int(x.argmax()) - max_lag
    if lag > 0:   est = est[lag:]
    elif lag < 0: ref = ref[-lag:]
    n = min(len(est), len(ref))
    return est[:n], ref[:n], lag / 16000.0

Whisper مع العربي بيرجّع أرقام عربية أو لاتينية بشكل غير ثابت، والتطويل (ـ) منتشر في الكتابة المصرية. الاتنين بيضخّموا WER بشكل مش حقيقي.

def norm(t):
    t = re.sub(r"[\u064B-\u0652\u0670\u0653-\u0655]", "", t)   # تشكيل كامل
    t = re.sub(r"[إأآٱ]", "ا", t).replace("ى", "ي").replace("ة", "ه")
    t = t.replace("\u0640", "")                                # تطويل
    t = t.translate(str.maketrans("٠١٢٣٤٥٦٧٨٩", "0123456789"))   # أرقام عربية
    t = re.sub(r"[^\w\s]", " ", t)
    return re.sub(r"\s+", " ", t).strip()
